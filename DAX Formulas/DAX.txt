Folder 01: Throughput & Workload
o	Total Task Created = countrows(‘fact_operational_request’)
o	Total Task Closed = calculate(countrows(‘fact_operational_request’[Request_Id]), ‘fact_operational_request’[Request_Status] = ‘Closed Complete’)
o	Total Task Cancelled = calculate(countrows(‘fact_operational_request’[Request_Id]), ‘fact_operational_request’[Request_Status] = ‘Cancelled’)
o	Active Open Backlog = CALCULATE(COUNTROWS('fact_operational_request'),'fact_operational_request'[Request_Status] IN { "Assigned", "Work In Progress", "Pending User Info" })
o	Monthly Target Achievement % = DIVIDE([Closed Tasks Throughput],800,0)
o	Monthly Target Variance = [Closed Tasks Throughput]-800
o	Net Queue Delta = [Total Request Ingested]-[Closed Tasks Throughput]-[Cancelled Tasks Count]    

Folder 02: SLA & Compliance
SLA Eligible Closed Tasks = CALCULATE(COUNTROWS(fact_operational_request), AND(fact_operational_request[Request_Status]="Closed Complete", fact_operational_request[SLA_Eligiblity]=1))

SLA Met Tasks = calculate(countrows(fact_operational_request), fact_operational_request[SLA_Status]="SLA Met")

SLA Breached Tasks = calculate(countrows(fact_operational_request), fact_operational_request[SLA_Status]="SLA Breached")

SLA Compliance % = DIVIDE([SLA Met Tasks],[SLA Eligible Closed Tasks],0)
